input_path = ./data_samples/my_documents

source_lang = auto
target_lang = en

# With source_lang = auto, the language used when detection cannot be trusted:
# the text is too short to judge (under LANG_ID_MIN_LETTERS letters), FastText is
# not confident enough (under LANG_ID_MIN_CONFIDENCE), or it names a language the
# translation backend cannot translate (e.g. 'yue', 'krc' on short Czech OCR
# fragments) — and neither the element's own label (ALTO LANG / xml:lang) nor the
# language detected for the whole document settles it. --default-source-lang on the
# CLI overrides this key; the DEFAULT_SOURCE_LANG env var applies only when this key is
# absent (comment it out to let the environment set it).
# Tip: on a single-language corpus, set source_lang = cs instead — no detection
# runs at all, and the CC BY-NC FastText model is not loaded.
default_source_lang = cs

# Use "xml,txt"  to process both AMCR .xml files and URL-list .txt files (the value
#                 without this key; a command line with --xpaths also gets it).
# Use "alto.xml,txt" when the input directory contains ALTO XML files and URL lists.
# Use "alto.xml"  when the input directory contains only ALTO XML files — the value
#                 shipped here; it switches ALTO mode on unless --xpaths or --formats
#                 is given on the command line.
formats = alto.xml

# Shipped samples are laid out as <mode>/<alto|xml>: in-place_translated_files/ holds
# replace-mode results, appended_translated_files/ append-mode ones (see
# data_samples/README.md). This default matches formats = alto.xml + output_mode = replace.
output = ./data_samples/in-place_translated_files/alto

fields = amcr-fields.txt

# Optional: path to a vocabulary CSV file (source_lemma,target_translation).
# When provided, the Tag-and-Protect strategy is activated:
# source terms are matched by lemma (via UDPipe), replaced with placeholder
# tags before translation, and restored with the vocabulary translations
# in the output.  Leave blank or comment out to disable.
vocabulary = data_samples/vocabulary.csv

# Translation backend selection (issue #4):
#   lindat             -> LINDAT CUBBITT (default; cs-centric, CC BY-NC-SA).
#   openai_compatible  -> free/low-cost OpenAI-compatible LLM API. Configure it
#                         with the LLM_* environment variables (LLM_BASE_URL,
#                         LLM_MODEL, LLM_API_KEY, optional LLM_PROVIDER /
#                         LLM_LANGUAGES). See docs/translation-backends.md.
#   ct2                -> self-hosted CTranslate2 model (EuroLLM / MADLAD-400 /
#                         NLLB-200 / Opus-MT). Needs requirements-ct2.txt and the
#                         CT2_* environment variables (CT2_MODEL_DIR at least).
# --backend on the CLI overrides this key; the TRANSLATION_BACKEND env var applies
# only when this key is absent.
translation_backend = lindat

# Output mode (issue #46) — how the translation is written into the document.
#   replace -> (default) overwrite the source-language field. The output is a
#              monolingual mirror of the input; the source text survives only in
#              the sidecar *_log.csv.
#   append  -> keep the source field and insert an xml:lang-marked sibling right
#              after it, the same shape AMCR's own thesaurus uses for
#              heslo / heslo_en. Re-running is then a no-op instead of a
#              double-translation, because the marker makes the pair detectable.
#              ALTO keeps every String's CONTENT (and geometry) and adds the
#              translation as <ALTERNATIVE PURPOSE="translation:<lang>">.
# --output-mode on the CLI overrides this key; the OUTPUT_MODE env var applies only
# when this key is absent.
# AMCR 2.2 accepts replace output and rejects append output (xml:lang is not declared
# on the free-text fields and they cannot repeat; checked with --xsd, issue #46): where
# an AMCR record must validate, use replace. ALTO append validates against ALTO 3.1.
output_mode = replace

# Backing service endpoints (atrium-project#63). Both are attachable: point them
# at a self-hosted UDPipe 2 / translation service, or a local stub, with no code
# change. They are environment variables, not keys in this file, because they
# vary per deployment rather than per run:
#   TRANSLATION_URL  -> translation API base URL, used by the lindat backend.
#                       LINDAT_BASE_URL is accepted as an alias.
#                       Default: https://lindat.mff.cuni.cz/services/translation/api/v2
#   UDPIPE_URL       -> UDPipe 2 endpoint used for vocabulary lemma matching.
#                       Same variable name as atrium-nlp-enrich, which calls the
#                       same service.
#                       Default: https://lindat.mff.cuni.cz/services/udpipe/api/process
# Unset, both reach the LINDAT hosts they always have. The LINDAT_* transport
# dials (LINDAT_MIN_INTERVAL_S, LINDAT_MAX_RETRIES, LINDAT_BACKOFF_BASE_S) are
# unrelated and keep their names.
